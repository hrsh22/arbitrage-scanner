   /// @notice Claim dust (amounts below MIN_CLAIM_THRESHOLD) after epoch finalization
   /// @param requestId The redemption request to claim dust from
   /// @param receiver Address to receive the assets
   /// @return assets Amount of dust assets transferred
   function claimDust(uint256 requestId, address receiver) 
       external nonReentrant returns (uint256 assets) {
       
       if (receiver == address(0)) revert InvalidAddress();
       
       RedemptionRequest storage request = redemptionRequests[requestId];
       if (!request.exists) revert InvalidRequest(requestId);
       if (request.controller != msg.sender) revert NotController(request.controller, msg.sender);
       if (request.status != RequestStatus.Claimable) revert RequestNotClaimable(requestId);
       
       // CRITICAL CHECK: Only allows dust claim when epoch is FINALIZED
       Epoch storage epoch = epochs[request.epochId];
       if (epoch.status != EpochStatus.Finalized) revert EpochNotFinalized(request.epochId);
       
       // Calculate remaining claimable assets for this request
       assets = request.assetsClaimable;
       
       // CRITICAL CHECK: Must be below threshold (this is dust)
       if (assets >= MIN_CLAIM_THRESHOLD) revert NoDustAvailable(requestId);
       if (assets == 0) revert ZeroAmount();
       
       uint256 shares = request.shares;
       
       // Update request - mark as claimed
       request.shares = 0;
       request.assetsClaimable = 0;
       request.status = RequestStatus.Claimed;
       
       // Transfer dust assets
       asset.safeTransfer(receiver, assets);
       
       emit DustClaimed(requestId, request.controller, receiver, assets, shares);
       
       return assets;
   }
