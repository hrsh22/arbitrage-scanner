export async function runDailyReconciliation(): Promise<void> {
  const epochs = await getActiveEpochs();
  
  const results = await Promise.all(
    epochs.map(epoch => reconcileEpochState(epoch.id, epoch.vaultAddress))
  );
  
  const failed = results.filter(r => !r.reconciled);
  
  if (failed.length > 0) {
    logger.error('Reconciliation failures detected', {
      count: failed.length,
      epochs: failed.map(r => r.epochId),
    });
    
    // Alert on-call
    await sendAlert({
      severity: 'critical',
      message: `${failed.length} epochs failed reconciliation`,
      details: failed,
    });
  } else {
    logger.info('Daily reconciliation passed', {
      epochsChecked: results.length,
    });
  }
}

function verifyConservation(state: EpochState): boolean {
  // cohortTotalAccrued == cohortTotalClaimed + cohortCarryRemaining
  const leftSide = state.cohortTotalClaimed + state.cohortCarryRemaining;
  const rightSide = state.cohortTotalAccrued;
  
  // Allow 10 wei tolerance for rounding
  const diff = leftSide > rightSide 
    ? leftSide - rightSide 
    : rightSide - leftSide;
  
  return diff <= 10n;
}
