export async function reconcileEpochState(
  epochId: number,
  vaultAddress: string
): Promise<ReconciliationResult> {
  const mismatches: Mismatch[] = [];
  
  // 1. Fetch contract state
  const contractState = await fetchContractEpochState(epochId, vaultAddress);
  
  // 2. Fetch repository state
  const repoState = await fetchRepositoryEpochState(epochId);
  
  // 3. Compare cohort totals
  if (contractState.cohortTotalAccrued !== repoState.cohortTotalAccrued) {
    mismatches.push({
      type: 'COHORT_ACCRUED_MISMATCH',
      field: 'cohortTotalAccrued',
      contractValue: contractState.cohortTotalAccrued,
      repoValue: repoState.cohortTotalAccrued,
      difference: contractState.cohortTotalAccrued - repoState.cohortTotalAccrued,
      epochId,
    });
  }
  
  if (contractState.cohortTotalClaimed !== repoState.cohortTotalClaimed) {
    mismatches.push({
      type: 'COHORT_CLAIMED_MISMATCH',
      field: 'cohortTotalClaimed',
      contractValue: contractState.cohortTotalClaimed,
      repoValue: repoState.cohortTotalClaimed,
      difference: contractState.cohortTotalClaimed - repoState.cohortTotalClaimed,
      epochId,
    });
  }
  
  if (contractState.cohortCarryRemaining !== repoState.cohortCarryRemaining) {
    mismatches.push({
      type: 'COHORT_CARRY_REMAINING_MISMATCH',
      field: 'cohortCarryRemaining',
      contractValue: contractState.cohortCarryRemaining,
      repoValue: repoState.cohortCarryRemaining,
      difference: contractState.cohortCarryRemaining - repoState.cohortCarryRemaining,
      epochId,
    });
  }
  
  // 4. Compare individual requests
  for (const contractRequest of contractState.requests) {
    const repoRequest = repoState.requests.find(
      r => r.requestId === contractRequest.requestId
    );
    
    if (!repoRequest) {
      mismatches.push({
        type: 'MISSING_REQUEST',
        requestId: contractRequest.requestId,
        epochId,
      });
      continue;
    }
    
    // Check request fields
    const fieldsToCompare = ['accrued', 'claimed', 'carryRemaining', 'status'] as const;
    for (const field of fieldsToCompare) {
      if (contractRequest[field] !== repoRequest[field]) {
        mismatches.push({
          type: 'REQUEST_FIELD_MISMATCH',
          requestId: contractRequest.requestId,
          field,
          contractValue: contractRequest[field],
          repoValue: repoRequest[field],
          epochId,
        });
      }
    }
  }
  
  // 5. Check for orphan repository records
  for (const repoRequest of repoState.requests) {
    const contractRequest = contractState.requests.find(
      r => r.requestId === repoRequest.requestId
    );
    if (!contractRequest) {
      mismatches.push({
        type: 'ORPHAN_REQUEST',
        requestId: repoRequest.requestId,
        epochId,
      });
    }
  }
  
  return {
    epochId,
    timestamp: new Date(),
    mismatches,
    reconciled: mismatches.length === 0,
  };
}

type MismatchType = 
  | 'COHORT_ACCRUED_MISMATCH'
  | 'COHORT_CLAIMED_MISMATCH'
  | 'COHORT_CARRY_REMAINING_MISMATCH'
  | 'REQUEST_FIELD_MISMATCH'
  | 'MISSING_REQUEST'
  | 'ORPHAN_REQUEST'
  | 'CONSERVATION_VIOLATION'
  | 'STATUS_MISMATCH';

interface Mismatch {
  type: MismatchType;
  epochId: number;
  requestId?: number;
  field?: string;
  contractValue?: bigint | string;
  repoValue?: bigint | string;
  difference?: bigint;
}

async function fetchContractEpochState(epochId: number, vaultAddress: string) {
  const epoch = await publicClient.readContract({
    address: vaultAddress as Hex,
    abi: EpochTrancheVaultABI,
    functionName: 'epochs',
    args: [BigInt(epochId)],
  });
  
  const requestIds = await publicClient.readContract({
    address: vaultAddress as Hex,
    abi: EpochTrancheVaultABI,
    functionName: 'getEpochRedemptionRequests',
    args: [BigInt(epochId)],
  });
  
  const requests = await Promise.all(
    requestIds.map(async (id) => {
      const req = await publicClient.readContract({
        address: vaultAddress as Hex,
        abi: EpochTrancheVaultABI,
        functionName: 'redemptionRequests',
        args: [id],
      });
      return { requestId: Number(id), ...req };
    })
  );
  
  return {
    cohortTotalAccrued: epoch.cohortTotalAccrued,
    cohortTotalClaimed: epoch.cohortTotalClaimed,
    cohortCarryRemaining: epoch.cohortCarryRemaining,
    requests,
  };
}

async function fetchRepositoryEpochState(epochId: number) {
  const requests = await db.query.redemptionRequests.findMany({
    where: eq(redemptionRequests.epochId, epochId),
  });
  
  const cohortState = await db.query.epochCohortState.findFirst({
    where: eq(epochCohortState.epochId, epochId),
  });
  
  return {
    cohortTotalAccrued: cohortState?.totalAccrued ?? 0n,
    cohortTotalClaimed: cohortState?.totalClaimed ?? 0n,
    cohortCarryRemaining: cohortState?.carryRemaining ?? 0n,
    requests,
  };
}

function generateReconciliationReport(result: ReconciliationResult): string {
  const lines = [
    `Reconciliation Report - Epoch ${result.epochId}`,
    `Timestamp: ${result.timestamp.toISOString()}`,
    `Status: ${result.reconciled ? 'RECONCILED' : 'MISMATCHES DETECTED'}`,
    `Mismatches: ${result.mismatches.length}`,
    '',
    ...result.mismatches.map(m => 
      `- ${m.type}: ${m.field || m.requestId || 'cohort'} ` +
      `(contract: ${m.contractValue}, repo: ${m.repoValue}, ` +
      `diff: ${m.difference || 'N/A'})`
    ),
  ];
  return lines.join('\n');
}
