```typescript
async settleEpoch(epochId: number, availableAssets: bigint, carryAmount: bigint): Promise<{
  success: boolean;
  txHash?: string;
  error?: string;
}> {
  try {
    // Validate preconditions
    const epochStatus = await this.getEpochStatus(epochId);
    if (epochStatus !== 'frozen') {
      return { 
        success: false, 
        error: `Epoch ${epochId} is not frozen (status: ${epochStatus})` 
      };
    }
    
    // Execute contract transaction
    const txHash = await this.client.settleEpoch({
      epochId: BigInt(epochId),
      availableAssets,
      carryAmount,
      walletClient: this.settlerWalletClient!,
    });
    
    // Wait for confirmation
    const receipt = await this.client.waitForTransactionReceipt(txHash);
    
    if (receipt.status !== 'success') {
      return { 
        success: false, 
        error: 'Transaction failed on-chain',
        txHash 
      };
    }
    
    // Persist tx hash for reconciliation
    await this.persistSettlementRecord({
      epochId,
      txHash,
      blockNumber: receipt.blockNumber,
      timestamp: new Date(),
    });
    
    return { success: true, txHash };
  } catch (error) {
    return { 
      success: false, 
      error: (error as Error).message 
    };
  }
}
```

```solidity
function settleEpoch(uint256 epochId, uint256 availableAssets, uint256 carryAmount)
    external
    onlyRole(SETTLER_ROLE)
    nonReentrant
{
    Epoch storage epoch = epochs[epochId];
    if (epoch.status != EpochStatus.Frozen && epoch.status != EpochStatus.Settling) {
        revert EpochNotFrozen(epochId);
    }
    
    // Calculate pro-rata distribution
    uint256 totalShares = epoch.frozenShares;
    uint256 totalAssets = availableAssets;
    uint256 proRataRatio = (totalAssets * PRORATA_PRECISION) / totalShares;
    
    // Process claims with chunked settlement
    uint256[] storage requests = epochRedemptionRequests[epochId];
    uint256 startIndex = nextRequestIndexToProcess[epochId];
    uint256 endIndex = startIndex + MAX_CHUNK_SIZE;
    if (endIndex > requests.length) endIndex = requests.length;
    
    for (uint256 i = startIndex; i < endIndex; i++) {
        RedemptionRequest storage request = redemptionRequests[requests[i]];
        if (!request.exists || request.status != RequestStatus.Frozen) continue;
        
        // Calculate claimable amount
        uint256 claimableShares = request.shares;
        uint256 claimableAssets = (claimableShares * totalAssets) / totalShares;
        uint256 carry = (claimableAssets * carryAmount) / PRORATA_PRECISION;
        
        // Update request state
        request.assetsClaimable = claimableAssets - carry;
        request.carryDeducted = carry;
        request.entitlement = claimableAssets;
        request.accrued = claimableAssets - carry;
        request.carryRemaining = claimableAssets - carry;
        request.status = RequestStatus.Claimable;
        
        // Update cohort totals
        epoch.cohortTotalEntitlement += claimableAssets;
        epoch.cohortTotalAccrued += claimableAssets - carry;
        epoch.cohortCarryRemaining += claimableAssets - carry;
    }
    
    // Update settlement progress
    nextRequestIndexToProcess[epochId] = endIndex;
    if (endIndex >= requests.length) {
        epoch.status = EpochStatus.Settled;
        epoch.carryAccrued = carryAmount;
        epoch.proRataRatio = proRataRatio;
    } else {
        epoch.status = EpochStatus.Settling;
    }
    
    emit EpochSettled(epochId, totalShares, totalAssets, carryAmount, proRataRatio);
}
```
