```solidity
function realizeLoss(uint256 epochId, uint256 lossAmount) 
    external 
    onlyRole(SETTLER_ROLE) 
    nonReentrant 
{
    Epoch storage epoch = epochs[epochId];
    if (!epoch.exists) revert InvalidRequest(epochId);
    if (epoch.status != EpochStatus.Settled && epoch.status != EpochStatus.Finalized) {
        revert EpochNotSettledOrFinalized(epochId);
    }
    if (lossAmount == 0) revert ZeroAmount();
    if (lossAmount > epoch.cohortCarryRemaining) {
        revert LossExceedsCarryRemaining(lossAmount, epoch.cohortCarryRemaining);
    }
    
    // Calculate total claimable shares for proportional distribution
    uint256[] storage requests = epochRedemptionRequests[epochId];
    uint256 totalClaimableShares;
    for (uint256 i = 0; i < requests.length; i++) {
        RedemptionRequest storage request = redemptionRequests[requests[i]];
        if (request.exists && request.status == RequestStatus.Claimable && request.shares > 0) {
            totalClaimableShares += request.shares;
        }
    }
    if (totalClaimableShares == 0) revert NoClaimableRequests(epochId);
    
    // Apply loss proportionally to each claimable request
    uint256 requestsAffected;
    for (uint256 i = 0; i < requests.length; i++) {
        RedemptionRequest storage request = redemptionRequests[requests[i]];
        if (request.exists && request.status == RequestStatus.Claimable && request.shares > 0) {
            uint256 requestLoss = (lossAmount * request.shares) / totalClaimableShares;
            
            // Reduce request accrued amount
            request.accrued -= requestLoss;
            request.carryRemaining -= requestLoss;
            requestsAffected++;
        }
    }
    
    // Update cohort totals
    epoch.cohortTotalAccrued -= lossAmount;
    epoch.cohortCarryRemaining -= lossAmount;
    
    emit LossRealized(epochId, lossAmount, requestsAffected);
}
```

--- LOSS ATTRIBUTION PROPERTIES ---
1. PROPORTIONAL: loss distributed by share proportion
   requestLoss = (totalLoss * request.shares) / totalClaimableShares

2. COHORT-ISOLATED: Only affects claimable requests in specified epoch
   Other epochs unaffected

3. BOUNDED: Cannot lose more than remaining carry
   lossAmount <= epoch.cohortCarryRemaining

4. CONSERVATION: Total loss equals sum of individual losses
   lossAmount ≈ SUM(requestLoss_i) for all i
   (minor rounding error acceptable)

--- LOSS FLOW ---
1. External event causes position loss (e.g., market movement)
2. Settler calls realizeLoss(epochId, lossAmount)
3. Contract validates:
   - Epoch is Settled or Finalized
   - Loss amount > 0
   - Loss amount <= cohortCarryRemaining
4. Contract calculates total claimable shares
5. For each claimable request:
   - Calculate proportional loss
   - Decrement request.accrued
   - Decrement request.carryRemaining
6. Update cohort totals
7. Emit LossRealized event

```solidity
function test_LossAttribution_Proportional() public {
    // Two users with equal shares
    address user2 = makeAddr("user2");
    _setupUser(user2, 100_000e6);
    
    _depositAndGetShares(user1, 100_000e6);
    _depositAndGetShares(user2, 100_000e6);
    
    uint256 shares1 = vault.balanceOf(user1);
    uint256 shares2 = vault.balanceOf(user2);
    
    _requestRedeem(user1, shares1);
    _requestRedeem(user2, shares2);
    
    uint256 epochId = vault.currentEpochId() - 1;
    _freezeEpoch();
    _settleEpoch(epochId, 0.1e18);
    
    uint256 accrued1Before = _getRequestAccrued(1);
    uint256 accrued2Before = _getRequestAccrued(2);
    
    uint256 cohortCarryRemaining = _getEpochCohortCarryRemaining(epochId);
    uint256 lossAmount = cohortCarryRemaining / 2;
    
    vm.prank(settler);
    vault.realizeLoss(epochId, lossAmount);
    
    // Equal shares = equal loss
    uint256 loss1 = accrued1Before - _getRequestAccrued(1);
    uint256 loss2 = accrued2Before - _getRequestAccrued(2);
    assertApproxEqRel(loss1, loss2, 0.01e18);
    assertEq(loss1 + loss2, lossAmount);
}
```
