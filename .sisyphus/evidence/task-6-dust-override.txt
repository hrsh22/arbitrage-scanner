```solidity
function claimDust(uint256 requestId) external nonReentrant returns (uint256 assets) {
    RedemptionRequest storage request = redemptionRequests[requestId];
    if (!request.exists) revert InvalidRequest(requestId);
    if (request.controller != msg.sender) revert NotController(request.controller, msg.sender);
    if (request.status != RequestStatus.Claimable) revert RequestNotClaimable(requestId);
    
    Epoch storage epoch = epochs[request.epochId];
    
    // CRITICAL: Only available after finalization
    if (epoch.status != EpochStatus.Finalized) revert EpochNotFinalized(request.epochId);
    
    assets = request.assetsClaimable;
    uint256 remainingCarry = request.carryRemaining;
    
    // Verify it's actually dust (below threshold)
    if (assets >= MIN_CLAIM_THRESHOLD) revert NotDust(assets, MIN_CLAIM_THRESHOLD);
    
    // Clear all remaining state
    request.shares = 0;
    request.assetsClaimable = 0;
    request.carryDeducted += remainingCarry;
    request.claimed += assets;
    request.carryRemaining = 0;
    request.status = RequestStatus.Claimed;
    epoch.cohortTotalClaimed += assets;
    epoch.cohortCarryRemaining -= remainingCarry;
    
    asset.safeTransfer(request.controller, assets);
    emit DustClaimed(requestId, request.controller, assets);
    return assets;
}
```

```solidity
function test_DustClaim_Succeeds_AfterFinalization() public {
    // Setup: Small deposit that results in dust after carry
    _depositAndGetShares(user1, 150e6);  // 150 USDC (just above threshold)
    uint256 shares = vault.balanceOf(user1);
    _requestRedeem(user1, shares);
    
    uint256 epochId = vault.currentEpochId() - 1;
    _freezeEpoch();
    _settleEpoch(epochId, 0.5e18);  // 50% carry reduces claimable to ~75 USDC
    
    uint256 requestId = 1;
    uint256 claimable = vault.getClaimableAssets(requestId);
    assertLt(claimable, MIN_CLAIM_THRESHOLD);  // Now dust
    
    // Cannot claim via normal redeem (blocked by threshold)
    vm.expectRevert(EpochTrancheVault.BelowClaimThreshold.selector);
    vault.redeem(requestId, shares, user1);
    
    // Finalize the epoch
    vm.prank(settler);
    vault.finalizeEpoch(epochId);
    
    // Now dust claim succeeds
    vm.prank(user1);
    uint256 claimed = vault.claimDust(requestId);
    
    assertEq(claimed, claimable);
    assertEq(vault.getClaimableAssets(requestId), 0);
}
```
