```typescript
async settleEpoch(
  epochId: number, 
  availableAssets: bigint, 
  carryAmount: bigint
): Promise<SettlementResult> {
  try {
    // 1. Pre-validation errors (before transaction)
    const validation = await this.validateSettlePreconditions(epochId);
    if (!validation.valid) {
      return {
        success: false,
        error: validation.reason,
        code: validation.errorCode,
      };
    }
    
    // 2. Transaction submission
    const txHash = await this.submitSettleTransaction({
      epochId,
      availableAssets,
      carryAmount,
    });
    
    // 3. Wait for confirmation with timeout
    const receipt = await Promise.race([
      this.waitForReceipt(txHash),
      this.settlementTimeout(),
    ]);
    
    // 4. Receipt status check
    if (!receipt) {
      return {
        success: false,
        error: 'Settlement transaction timed out',
        txHash,
        code: 'TIMEOUT',
      };
    }
    
    if (receipt.status === 'reverted') {
      return {
        success: false,
        error: 'Transaction reverted on-chain',
        txHash,
        code: 'REVERTED',
      };
    }
    
    return { success: true, txHash };
    
  } catch (error) {
    // 5. Unexpected errors
    return {
      success: false,
      error: (error as Error).message,
      code: 'UNEXPECTED_ERROR',
    };
  }
}
```
